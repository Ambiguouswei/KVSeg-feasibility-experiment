cmake_minimum_required(VERSION 3.22)
project(kvcohort VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The prototype requires Linux file descriptors, flock and fsync")
endif()
option(KVCOHORT_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_path(ROCKSDB_INCLUDE_DIR rocksdb/db.h REQUIRED)
find_library(ROCKSDB_LIBRARY NAMES rocksdb REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
add_library(kvcohort_core src/store.cpp src/policy.cpp src/scheduler.cpp)
target_include_directories(kvcohort_core PUBLIC include PRIVATE ${ROCKSDB_INCLUDE_DIR})
target_link_libraries(kvcohort_core PUBLIC ${ROCKSDB_LIBRARY} OpenSSL::Crypto Threads::Threads)
target_compile_options(kvcohort_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(KVCOHORT_SANITIZE)
  target_compile_options(kvcohort_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(kvcohort_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(kvc src/main.cpp)
target_link_libraries(kvc PRIVATE kvcohort_core)
add_executable(kvc_replay src/replay.cpp)
target_link_libraries(kvc_replay PRIVATE kvcohort_core)
target_compile_options(kvc_replay PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)

# 客户端库：把 Unix 域套接字协议封装成与进程内 Store 同形的调用。
add_library(kvcohort_client src/client.cpp)
target_include_directories(kvcohort_client PUBLIC include)
target_link_libraries(kvcohort_client PUBLIC kvcohort_core)
target_compile_options(kvcohort_client PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)

# 守护进程：独占数据根，对外提供服务。
add_executable(kvd src/server.cpp)
target_link_libraries(kvd PRIVATE kvcohort_core)
target_compile_options(kvd PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
include(CTest)
if(BUILD_TESTING)
  add_executable(kvc_tests tests/store_test.cpp)
  target_link_libraries(kvc_tests PRIVATE kvcohort_core)
  add_test(NAME storage_contract COMMAND kvc_tests)
  set_tests_properties(storage_contract PROPERTIES TIMEOUT 60 ENVIRONMENT "KVCOHORT_TEST_ROOT=${CMAKE_SOURCE_DIR}/var/tests")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME replay_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/replay_test.py $<TARGET_FILE:kvc_replay>)
  set_tests_properties(replay_contract PROPERTIES TIMEOUT 60 ENVIRONMENT "KVCOHORT_TEST_ROOT=${CMAKE_SOURCE_DIR}/var/tests")
  # 服务端协议的独立实现（Python）交叉验证，避免"自己测自己"。
  add_test(NAME daemon_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/daemon_test.py $<TARGET_FILE:kvd>)
  set_tests_properties(daemon_contract PROPERTIES TIMEOUT 180)

  add_executable(scheduler_test tests/scheduler_test.cpp)
  target_link_libraries(scheduler_test PRIVATE kvcohort_core)
  target_compile_options(scheduler_test PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  add_test(NAME scheduler_contract COMMAND scheduler_test)
  set_tests_properties(scheduler_contract PROPERTIES TIMEOUT 120)
endif()
